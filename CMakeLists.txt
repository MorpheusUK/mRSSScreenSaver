cmake_minimum_required(VERSION 3.24)
project(mRSSScreenSaver VERSION 1.0.0 LANGUAGES C CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This project requires 64-bit Windows.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(PCRE2_BUILD_PCRE2_8 OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_16 ON CACHE BOOL "" FORCE)
set(PCRE2_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2GREP OFF CACHE BOOL "" FORCE)
set(PCRE2_STATIC_PIC ON CACHE BOOL "" FORCE)
include(FetchContent)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/pcre2/CMakeLists.txt")
  FetchContent_Declare(pcre2_source GIT_REPOSITORY https://github.com/PCRE2Project/pcre2.git
    GIT_TAG b2bd4254b379b9d7dc9a3dda060a7e27009ccdff SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/pcre2")
  FetchContent_GetProperties(pcre2_source)
  FetchContent_Populate(pcre2_source)
endif()
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/json/single_include/nlohmann/json.hpp")
  FetchContent_Declare(json_source GIT_REPOSITORY https://github.com/nlohmann/json.git
    GIT_TAG 55f93686c01528224f448c19128836e7df245f72 SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/json")
  FetchContent_GetProperties(json_source)
  FetchContent_Populate(json_source)
endif()
add_subdirectory(third_party/pcre2 EXCLUDE_FROM_ALL)
add_library(mrss_core src/policy.cpp src/feed.cpp src/json_feed.cpp src/settings.cpp src/filter.cpp src/network.cpp src/engine.cpp)
target_include_directories(mrss_core PUBLIC include)
target_include_directories(mrss_core PRIVATE "${CMAKE_BINARY_DIR}/generated")
target_include_directories(mrss_core SYSTEM PUBLIC third_party/json/single_include)
target_compile_definitions(mrss_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN PCRE2_STATIC NTDDI_VERSION=0x0A00000C _WIN32_WINNT=0x0A00)
target_compile_options(mrss_core PRIVATE /W4 /WX /permissive- /utf-8 /sdl /guard:cf)
target_link_libraries(mrss_core PUBLIC winhttp xmllite shlwapi ole32 shell32 bcrypt ws2_32 iphlpapi pcre2-16-static)
set(DARKMODELIB_ROOT "I:/darkmodelib" CACHE PATH "Path to pinned darkmodelib source")
if(NOT EXISTS "${DARKMODELIB_ROOT}/include/Darkmodelib.h")
  message(FATAL_ERROR "Set DARKMODELIB_ROOT to the darkmodelib checkout")
endif()
file(GLOB DARKMODELIB_SOURCES CONFIGURE_DEPENDS "${DARKMODELIB_ROOT}/src/*.cpp")
add_library(darkmodelib STATIC ${DARKMODELIB_SOURCES})
target_include_directories(darkmodelib PUBLIC "${DARKMODELIB_ROOT}/include")
target_compile_definitions(darkmodelib PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN NTDDI_VERSION=0x0A00000C _WIN32_WINNT=0x0A00 PRIVATE _DARKMODELIB_NO_INI_CONFIG _DARKMODELIB_CUSTOM_MEM=0x002)
target_compile_options(darkmodelib PRIVATE /utf-8 /Zc:__cplusplus)
add_library(mrss_graphics src/graphics.cpp)
target_link_libraries(mrss_graphics PUBLIC mrss_core d3d11 dxgi dxguid d2d1 dwrite windowscodecs propsys)
add_executable(mrss_probe src/probe.cpp)
target_link_libraries(mrss_probe PRIVATE mrss_core)
add_executable(mrss WIN32 src/main.cpp src/app.rc)
configure_file(src/version.hpp.in "${CMAKE_BINARY_DIR}/generated/mrss/version.hpp" @ONLY)
target_include_directories(mrss PRIVATE "${CMAKE_BINARY_DIR}/generated")
set_target_properties(mrss PROPERTIES OUTPUT_NAME "mRSSScreenSaver" SUFFIX ".scr" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
target_link_libraries(mrss PRIVATE mrss_graphics darkmodelib comctl32 comdlg32 powrprof)
target_compile_options(mrss PRIVATE /W4 /permissive- /utf-8 /sdl /guard:cf)
target_link_options(mrss PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /CETCOMPAT /MANIFEST:NO)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/package/licenses")
configure_file(docs/TESTING.md "${CMAKE_BINARY_DIR}/package/TESTING.md" COPYONLY)
configure_file(docs/DEPENDENCIES.md "${CMAKE_BINARY_DIR}/package/DEPENDENCIES.md" COPYONLY)
configure_file(LICENSE "${CMAKE_BINARY_DIR}/package/LICENSE.txt" COPYONLY)
configure_file(NOTICE.md "${CMAKE_BINARY_DIR}/package/NOTICE.md" COPYONLY)
file(COPY "${DARKMODELIB_ROOT}/LICENSE.md" "${DARKMODELIB_ROOT}/LICENSE-MIT.md" DESTINATION "${CMAKE_BINARY_DIR}/package/licenses/darkmodelib")
file(COPY "${DARKMODELIB_ROOT}/include" "${DARKMODELIB_ROOT}/src" DESTINATION "${CMAKE_BINARY_DIR}/package/licenses/darkmodelib/source")
file(COPY "third_party/pcre2/LICENCE.md" DESTINATION "${CMAKE_BINARY_DIR}/package/licenses/pcre2")
file(COPY "third_party/json/LICENSE.MIT" DESTINATION "${CMAKE_BINARY_DIR}/package/licenses/json")
find_program(INNO_COMPILER ISCC HINTS "C:/Program Files/Inno Setup 7")
if(INNO_COMPILER)
  add_custom_target(installer
    COMMAND "${INNO_COMPILER}" "/DBuildRoot=${CMAKE_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/installer/mRSSScreenSaver.iss"
    DEPENDS mrss VERBATIM)
endif()
include(CTest)
if(BUILD_TESTING)
  set(gtest_force_shared_crt OFF CACHE BOOL "" FORCE)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/googletest/CMakeLists.txt")
    add_subdirectory(third_party/googletest EXCLUDE_FROM_ALL)
  else()
    include(FetchContent)
    FetchContent_Declare(googletest
      GIT_REPOSITORY https://github.com/google/googletest.git
      GIT_TAG 52eb8108c5bdec04579160ae17225d66034bd723)
    FetchContent_MakeAvailable(googletest)
  endif()
  add_executable(mrss_tests tests/policy_tests.cpp tests/feed_tests.cpp tests/integration_tests.cpp)
  target_link_libraries(mrss_tests PRIVATE mrss_graphics GTest::gtest_main)
  target_compile_options(mrss_tests PRIVATE /W4 /WX /permissive- /utf-8 /sdl)
  include(GoogleTest)
  gtest_discover_tests(mrss_tests)
  add_custom_target(check COMMAND "${CMAKE_CTEST_COMMAND}" -C $<CONFIG> --output-on-failure
    WORKING_DIRECTORY "${CMAKE_BINARY_DIR}" DEPENDS mrss_tests VERBATIM)
  if(TARGET installer)
    add_dependencies(installer check)
  endif()
endif()
